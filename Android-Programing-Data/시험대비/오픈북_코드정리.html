Java 뼈대

package com.example.chapter07;   // ← 바꾸기: 내 프로젝트 패키지명 그대로 둘 것

import android.os.Bundle;
import android.view.View;
import android.widget.*;          // Button, EditText, TextView, ImageView, Toast ... 전부
import android.graphics.Color;
import androidx.appcompat.app.AppCompatActivity;

public class Self0701 extends AppCompatActivity {     // ← 바꾸기: 파일 이름과 같게

    // 1) 위젯 변수 선언 (클래스 안, 메서드 밖)
    Button button1;
    EditText edit1;
    TextView textView1;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.self0701_main);    // ← 바꾸기: XML 파일 이름
        setTitle("제목");                            // ← 바꾸기: 앱 상단 제목

        // 2) XML의 위젯과 연결
        button1 = (Button) findViewById(R.id.button1);
        edit1 = (EditText) findViewById(R.id.edit1);
        textView1 = (TextView) findViewById(R.id.textView1);

        // 3) 이벤트 처리
        button1.setOnClickListener(new View.OnClickListener() {
            public void onClick(View v) {
                // 버튼 눌렀을 때 할 일
            }
        });
    }
}

import android.widget.*; 하나로 위젯 import를 한 번에 해결. 빨간 글씨가 뜨면 그 단어에 커서 → Alt + Enter (Mac: Option + Enter)

값 읽기 / 쓰기 / 변환

하고 싶은 것 코드
EditText 글자 읽기 String s = edit1.getText().toString();
비어 있는지 검사 if (s.trim().equals("")) { ... }
글자 → 정수 int n = Integer.parseInt(s);
글자 → 실수 double d = Double.parseDouble(s);
글자 → float (회전 각도 등) float f = Float.parseFloat(s);
정수 → 글자 (setText에 넣을 때 필수!) Integer.toString(n) 또는 "" + n
실수 → 글자 Double.toString(d) / Double 객체면 result.toString()
TextView에 출력 textView1.setText("결과 : " + result);
EditText 비우기 edit1.setText("");
소수점 첫째 자리까지 자르기 result = (int)(result * 10) / 10.0;

같은 동작 버튼 여러 개 (배열 + for문)

Java
Button[] numButtons = new Button[10];
Integer[] numBtnIDs = { R.id.BtnNum0, R.id.BtnNum1, R.id.BtnNum2 /* ... */ };

for (int i = 0; i < numBtnIDs.length; i++) {
    numButtons[i] = (Button) findViewById(numBtnIDs[i]);
}
for (int i = 0; i < numBtnIDs.length; i++) {
    final int index = i;              // 익명 클래스에서 쓰려면 final
    numButtons[index].setOnClickListener(new View.OnClickListener() {
        public void onClick(View v) {
            edit1.setText(edit1.getText().toString()
                    + numButtons[index].getText().toString());
        }
    });
}

XML 속성 사전 4장

속성 의미 값 예시
layout_width / layout_height 크기 match_parent(부모만큼), wrap_content(내용만큼), 100dp
id 이름 (Java에서 찾을 때) @+id/button1
text / hint 글자 / 흐린 안내 글자
textSize 글자 크기 20sp
textColor 글자색 #FF0000
textStyle 굵게/기울임 bold, italic, bold|italic
background 배경색 (Button은 backgroundTint 사용) #00ff00
layout_margin 바깥 여백 20dp
padding 안쪽 여백 20dp
visibility 보이기 visible, invisible, gone
enabled 사용 가능 false → 비활성
clickable 클릭 가능 false
rotation 회전 45
singleLine 한 줄 입력 true
inputType 입력 종류 number, numberDecimal, phone, textPassword
checked 처음부터 체크 true
src 그림 @drawable/jeju2
scaleType 그림 크기 맞춤 fitXY, fitCenter, center

색 표기: #RRGGBB 또는 #AARRGGBB (AA = 투명도, FF = 불투명)
